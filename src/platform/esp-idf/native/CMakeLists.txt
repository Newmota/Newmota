cmake_minimum_required(VERSION 3.16)

if(NOT NEWMOTA_ROOT)
  message(FATAL_ERROR "NEWMOTA_ROOT is required")
endif()
if(NOT NEWMOTA_CONSOLE)
  message(FATAL_ERROR "NEWMOTA_CONSOLE is required")
endif()
if(NOT NEWMOTA_IMAGE_ROLE)
  message(FATAL_ERROR "NEWMOTA_IMAGE_ROLE is required")
endif()
if(NOT NEWMOTA_BUILD_DATE OR NOT NEWMOTA_BUILD_TIME)
  message(FATAL_ERROR "NEWMOTA_BUILD_DATE and NEWMOTA_BUILD_TIME are required")
endif()
if(NOT NEWMOTA_VERSION)
  message(FATAL_ERROR "NEWMOTA_VERSION is required")
endif()
if(NOT NEWMOTA_ESP_LAYOUT_ID OR NOT NEWMOTA_ESP_APP0_OFFSET OR
   NOT NEWMOTA_ESP_SAFEBOOT_OFFSET OR NOT NEWMOTA_ESP_OTADATA_OFFSET OR
   NOT NEWMOTA_ESP_NVS_OFFSET OR NOT NEWMOTA_ESP_SDK_TARGET)
  message(FATAL_ERROR "NEWMOTA ESP image compatibility metadata is required")
endif()
if(NOT NEWMOTA_FEATURE_DIR)
  message(FATAL_ERROR "NEWMOTA_FEATURE_DIR is required")
endif()
if(NOT NEWMOTA_HARDWARE_DIR)
  message(FATAL_ERROR "NEWMOTA_HARDWARE_DIR is required")
endif()
if(NOT NEWMOTA_BERRY_GENERATED_DIR)
  message(FATAL_ERROR "NEWMOTA_BERRY_GENERATED_DIR is required")
endif()

# ESP-Hosted and, before IDF 6.1, esp_wifi_remote are vendored IDF components.
include("${NEWMOTA_FEATURE_DIR}/requirements.cmake")
if(NEWMOTA_USE_ESP_HOSTED)
  list(APPEND EXTRA_COMPONENT_DIRS "${NEWMOTA_ROOT}/src/third_party/esp_hosted")
  if(NOT EXISTS "$ENV{IDF_PATH}/components/esp_wifi/remote/Kconfig.wifi.in")
    list(APPEND EXTRA_COMPONENT_DIRS "${NEWMOTA_ROOT}/src/third_party/esp_wifi_remote")
  endif()
endif()

set(PROJECT_VER "${NEWMOTA_VERSION}")
# ESP-IDF resets the policy version while importing ExternalProject. Adopt its
# build-local install prefix even though the bootloader has no install step.
if(POLICY CMP0225)
  set(CMAKE_POLICY_DEFAULT_CMP0225 NEW)
endif()
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
idf_build_set_property(MINIMAL_BUILD ON)
idf_build_set_property(NEWMOTA_CONSOLE "${NEWMOTA_CONSOLE}")
idf_build_set_property(NEWMOTA_IMAGE_ROLE "${NEWMOTA_IMAGE_ROLE}")
idf_build_set_property(NEWMOTA_BUILD_DATE "${NEWMOTA_BUILD_DATE}")
idf_build_set_property(NEWMOTA_BUILD_TIME "${NEWMOTA_BUILD_TIME}")
idf_build_set_property(NEWMOTA_VERSION "${NEWMOTA_VERSION}")
idf_build_set_property(NEWMOTA_ESP_LAYOUT_ID "${NEWMOTA_ESP_LAYOUT_ID}")
idf_build_set_property(NEWMOTA_ESP_APP0_OFFSET "${NEWMOTA_ESP_APP0_OFFSET}")
idf_build_set_property(NEWMOTA_ESP_SAFEBOOT_OFFSET "${NEWMOTA_ESP_SAFEBOOT_OFFSET}")
idf_build_set_property(NEWMOTA_ESP_OTADATA_OFFSET "${NEWMOTA_ESP_OTADATA_OFFSET}")
idf_build_set_property(NEWMOTA_ESP_NVS_OFFSET "${NEWMOTA_ESP_NVS_OFFSET}")
idf_build_set_property(NEWMOTA_ESP_PSRAM_BYTES "${NEWMOTA_ESP_PSRAM_BYTES}")
idf_build_set_property(NEWMOTA_ESP_SDK_TARGET "${NEWMOTA_ESP_SDK_TARGET}")
idf_build_set_property(NEWMOTA_FEATURE_DIR "${NEWMOTA_FEATURE_DIR}")
idf_build_set_property(NEWMOTA_HARDWARE_DIR "${NEWMOTA_HARDWARE_DIR}")
idf_build_set_property(NEWMOTA_BERRY_GENERATED_DIR "${NEWMOTA_BERRY_GENERATED_DIR}")
project(newmota)
